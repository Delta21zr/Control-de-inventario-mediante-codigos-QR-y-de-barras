    <script>
        const productos = [
            { id: 'PRD001', nombre: 'Mouse inalámbrico', stock: 10 },
            { id: 'PRD002', nombre: 'Teclado USB', stock: 5 },
            { id: 'PRD003', nombre: 'Cable HDMI', stock: 12 }
        ];

        const container = document.getElementById('cards-container');

        productos.forEach((prod) => {
            const card = document.createElement('div');
            card.className = 'card';
            const info = document.createElement('div');
            info.innerHTML = `<h3>${prod.nombre}</h3><p><strong>CÓDIGO:</strong> ${prod.id} <br> <strong>Stock:</strong>${prod.stock}</p>`;
            card.appendChild(info);

            const qrDiv = document.createElement('div');
            qrDiv.className = 'qr-container';
            const qrElement = document.createElement('div');
            qrDiv.appendChild(qrElement);
            card.appendChild(qrDiv);

            new QRCode(qrElement, {
                text: prod.id, width: 160, height: 160, colorDark : "#000000", colorLight : "#ffffff", correctLevel : QRCode.CorrectLevel.H
            });

            const barcodeDiv = document.createElement('div');
            barcodeDiv.className = 'barcode-container';
            const barcodeSvg = document.createElement('svg');
            barcodeSvg.id = `barcode-${prod.id}`;
            barcodeDiv.appendChild(barcodeSvg);
            card.appendChild(barcodeDiv);
            container.appendChild(card);

            JsBarcode(`#barcode-${prod.id}`, prod.id, {
                format: "CODE128", width: 2.5, height: 70, displayValue: true, fontSize: 20, fontOptions: "bold", textMargin: 8, background: "#ffffff", lineColor: "#000000"
            });
        });
    </script>
