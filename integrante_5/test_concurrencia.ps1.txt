$url = "http://192.168.1.X/inventarioqr/api/movimientos.php"
$payload = '{"codigo":"PRD001","tipo":"salida","cantidad":4}'

Write-Host "Iniciando prueba de concurrencia..." -ForegroundColor Cyan
Write-Host "URL Destino: $url"
Write-Host "Payload: $payload"
Write-Host "Enviando 2 peticiones simultáneas...`n"

$scriptBlock = {
    param($url,$payload, $hiloId)$headers = @{ "Content-Type" = "application/json" }
    try {
        $response = Invoke-WebRequest -Uri$url -Method Post -Body $payload -Headers$headers -UseBasicParsing -ErrorAction Stop
        return [PSCustomObject]@{ Hilo = $hiloId; EstadoHTTP = $response.StatusCode; Respuesta =$response.Content; Resultado = "ACEPTADA" }
    }
    catch {
        $statusCode = [int]$_.Exception.Response.StatusCode
        $stream = $_.Exception.Response.GetResponseStream()$reader = New-Object System.IO.StreamReader($stream)$responseBody = $reader.ReadToEnd()$reader.Close()
        return [PSCustomObject]@{ Hilo = $hiloId; EstadoHTTP = $statusCode; Respuesta =$responseBody; Resultado = "RECHAZADA" }
    }
}

$job1 = Start-Job -ScriptBlock$scriptBlock -ArgumentList $url,$payload, "Petición A"
$job2 = Start-Job -ScriptBlock$scriptBlock -ArgumentList $url,$payload, "Petición B"

Wait-Job -Job $job1,$job2 | Out-Null

$resultado1 = Receive-Job -Job$job1
$resultado2 = Receive-Job -Job$job2

Remove-Job -Job $job1,$job2

function Imprimir-Resultado ($res) {
    if ($res.Resultado -eq "ACEPTADA") {
        Write-Host "[$($res.Hilo)] - HTTP $($res.EstadoHTTP) | $($res.Resultado)" -ForegroundColor Green
        Write-Host "Respuesta: $($res.Respuesta)`n" -ForegroundColor Gray
    } else {
        Write-Host "[$($res.Hilo)] - HTTP $($res.EstadoHTTP) | $($res.Resultado)" -ForegroundColor Red
        Write-Host "Respuesta: $($res.Respuesta)`n" -ForegroundColor Gray
    }
}

Imprimir-Resultado $resultado1
Imprimir-Resultado $resultado2

Write-Host "Prueba finalizada. Validar en BD que el stock haya quedado en 1." -ForegroundColor Yellow
